cmake_minimum_required(VERSION 3.16)
project(saarathi_native_server LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_path(BOOST_HEADERS boost/beast.hpp REQUIRED)
add_executable(saarathi_server server.cpp)
target_include_directories(saarathi_server PRIVATE ${BOOST_HEADERS})
target_compile_definitions(saarathi_server PRIVATE BOOST_ERROR_CODE_HEADER_ONLY BOOST_JSON_NO_LIB)
target_link_libraries(saarathi_server PRIVATE Threads::Threads)
if(APPLE)
  target_compile_options(saarathi_server PRIVATE -Wall -Wextra -Wpedantic)
else()
  target_compile_options(saarathi_server PRIVATE -Wall -Wextra -Wpedantic)
endif()
